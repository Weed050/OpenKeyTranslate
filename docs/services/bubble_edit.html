services.bubble_edit

Manual bubble operations on a page's saved JSON (the editor's "Re-OCR bubble", "Add bubble", "Merge" - any number of bubbles, picked with Shift+click on the page - and "Split" - lines of one bubble into two, translation divided): pure functions on the ocr_data dict + image + an OCR function, so they are testable without a server.

JSON shape they work with (services/page_export.save_ocr_json): bubbles[]: {bubble_id, text, translation, ai_translation, box (4 points), line_count, avg_score, [text_raw, skip, skip_reason, interior_box, missed_text, refined]} lines[]: {line_id, box, bubble_id} All coordinates are original-image coordinates.

 39def _next_number(ids, prefix: str) -> int:
 40    nums = [int(i.split("_")[-1]) for i in ids if i.startswith(prefix) and i.split("_")[-1].isdigit()]

 44def _replace_lines(ocr_data: dict, bubble_id: str, new_lines: list[dict]) -> None:
 45    if "lines" not in ocr_data:
 46        return       # page processed before line data existed: creating a partial list would make a rebuild bring the old text back
 47    lines = ocr_data["lines"]
 48    base = _next_number([l["line_id"] for l in lines], "line_")
 49    kept = [l for l in lines if l.get("bubble_id") != bubble_id]
 50    for k, l in enumerate(new_lines):
 51        kept.append({"line_id": f"line_{base + k}", "box": [[round(x, 1), round(y, 1)] for x, y in l["box"]], "bubble_id": bubble_id,
 52                     "text": l.get("text", ""), **({"angle": l["angle"]} if "angle" in l else {})})
 53    ocr_data["lines"] = kept

 81def reocr_bubble(ocr_data: dict, bubble_id: str, image, ocr_region_fn, protected=(), box=None) -> dict:
 82    """
 83    Read the bubble's whole region again as one crop and replace its text + lines. The old text goes to text_raw.
 84    `box` = [x1, y1, x2, y2] drawn by the user: that exact area is read (for a bubble whose detected box misses a line,
 85    e.g. text below the box); without it the region is the bubble interior (flood fill) or the bubble box + margin.
 86    """
 87    bubble = _find(ocr_data, bubble_id)
 88    if box is not None:
 89        ih, iw = image.shape[:2]
 90        x1, y1, x2, y2 = (float(v) for v in box)
 91        x1, x2 = sorted((max(0.0, x1), min(float(iw), x2)))
 92        y1, y2 = sorted((max(0.0, y1), min(float(ih), y2)))
 93        if x2 - x1 < 12 or y2 - y1 < 12:
 94            raise BubbleEditError("The selected area is too small.", 400)
 95        region = (int(x1), int(y1), int(x2), int(y2))
 96    else:
 97        region = _reading_region(image, bubble)
 98    reading = read_region(image, region, ocr_region_fn)
 99    if reading is None:
100        raise BubbleEditError("No text was found in this bubble's area.", 422)
101
102    new_text = _clean(reading["text"], protected)
103    changed = new_text != bubble["text"]
104    if changed:
105        bubble.setdefault("text_raw", bubble["text"])
106        bubble["text"] = new_text
107    bubble.update(box=[[round(x, 1), round(y, 1)] for x, y in reading["box"]], line_count=len(reading["lines"]),
108                  avg_score=round(reading["avg_score"], 4))
109    bubble.pop("refined", None)
110    _replace_lines(ocr_data, bubble_id, reading["lines"])
111    reason = skip_reason(bubble["text"], skip_symbols=get_setting("ocr_skip_symbols", True))
112    if reason and not (reason == "number" and not get_setting("ocr_skip_numeric", True)) and not bubble.get("skip"):
113        bubble["skip"], bubble["skip_reason"] = True, reason
114    _refresh_info(image, ocr_data, bubble)
115    return {"bubble": bubble, "text_changed": changed}

118def add_bubble(ocr_data: dict, box, image, ocr_region_fn, protected=()) -> dict:
119    """OCR a rectangle the user drew and add its content as a NEW bubble (text OCR never saw)."""
120    ih, iw = image.shape[:2]
121    x1, y1, x2, y2 = (float(v) for v in box)
122    x1, x2 = sorted((max(0.0, x1), min(float(iw), x2)))
123    y1, y2 = sorted((max(0.0, y1), min(float(ih), y2)))
124    if x2 - x1 < 12 or y2 - y1 < 12:
125        raise BubbleEditError("The selected area is too small.", 400)
126    reading = read_region(image, (x1, y1, x2, y2), ocr_region_fn)
127    if reading is None:
128        raise BubbleEditError("No text was found in the selected area.", 422)
129
130    bubbles = ocr_data.setdefault("bubbles", [])
131    bubble_id = f"bubble_{_next_number([b['bubble_id'] for b in bubbles], 'bubble_')}"
132    text = _clean(reading["text"], protected)
133    bubble = {
134        "bubble_id": bubble_id, "text": text, "translation": "", "ai_translation": "",
135        "box": [[round(x, 1), round(y, 1)] for x, y in reading["box"]], "line_count": len(reading["lines"]),
136        "avg_score": round(reading["avg_score"], 4), "added_by_user": True,
137    }
138    if text != reading["text"]:
139        bubble["text_raw"] = reading["text"]
140    reason = skip_reason(text, skip_symbols=get_setting("ocr_skip_symbols", True))
141    if reason and not (reason == "number" and not get_setting("ocr_skip_numeric", True)):
142        bubble["skip"], bubble["skip_reason"] = True, reason
143
144    cy = center(bbox_of(bubble["box"]))[1]
145    index = next((i for i, b in enumerate(bubbles) if center(bbox_of(b["box"]))[1] > cy), len(bubbles))   # keep top-to-bottom order
146    bubbles.insert(index, bubble)
147    _replace_lines(ocr_data, bubble_id, reading["lines"])
148    _refresh_info(image, ocr_data, bubble)
149    return {"bubble": bubble, "index": index}

152def _merge_parts(ocr_data: dict, bubble_id: str, other_ids, order=None) -> list[dict]:
153    """
154    The bubbles to merge, in the order their texts are joined. Default = the page's reading order (page JSON
155    "bubble_order", else the setting bubble_order: ttb / rtl / ltr) applied to just these pieces, so two pieces side by
156    side in a right-to-left manga join right piece first. `order` (every id exactly once, from the editor's merge
157    preview) overrides it.
158    """
159    others = [other_ids] if isinstance(other_ids, str) else list(other_ids or [])
160    ids = [bubble_id] + [i for i in dict.fromkeys(others) if i != bubble_id]
161    if len(ids) < 2:
162        raise BubbleEditError("Choose at least two different bubbles.", 400)
163    parts = {i: _find(ocr_data, i) for i in ids}
164    if order:
165        if sorted(order) != sorted(ids):
166            raise BubbleEditError("order must list every merged bubble exactly once", 400)
167        return [parts[i] for i in order]
168    return auto_order(list(parts.values()), ocr_data.get("bubble_order") or get_setting("bubble_order", "ttb"))

182def merge_preview(ocr_data: dict, bubble_id: str, other_ids, order=None) -> dict:
183    """
184    What merge_bubbles would produce, without changing anything (the editor's merge dialog). `covered` = bubbles NOT
185    being merged whose centre lies inside the merged box - the merged translation would be typeset over them.
186    """
187    parts = _merge_parts(ocr_data, bubble_id, other_ids, order)
188    ids = [p["bubble_id"] for p in parts]
189    x1, y1, x2, y2 = _union_of(parts)
190    covered = [b["bubble_id"] for b in ocr_data.get("bubbles", []) if b["bubble_id"] not in ids
191               and x1 <= center(bbox_of(b["box"]))[0] <= x2 and y1 <= center(bbox_of(b["box"]))[1] <= y2]
192    return {"order": ids, "text": _joined(parts, "text"), "translation": _joined(parts, "translation"), "covered": covered}

195def merge_bubbles(ocr_data: dict, bubble_id: str, other_ids, image=None, order=None) -> dict:
196    """
197    Merge one or more bubbles INTO `bubble_id` (one speech bubble the OCR cut into pieces; the pieces need not be
198    neighbours in the list). Texts and translations are joined in reading order (see _merge_parts). The merged bubble
199    keeps `bubble_id` and takes the earliest list position of the pieces. Its "reviewed" stamp is dropped: the joined
200    translation is new text nobody has confirmed yet.
201    """
202    parts = _merge_parts(ocr_data, bubble_id, other_ids, order)
203    a = _find(ocr_data, bubble_id)
204    removed = [p["bubble_id"] for p in parts if p["bubble_id"] != bubble_id]
205    first_index = min(i for i, x in enumerate(ocr_data["bubbles"]) if x["bubble_id"] in removed or x is a)
206
207    text_raw = (" ".join((p.get("text_raw") or p.get("text") or "").strip() for p in parts).strip()
208                if any(p.get("text_raw") for p in parts) else None)      # a piece without text_raw: its text IS the raw OCR
209    a.update(text=_joined(parts, "text"), translation=_joined(parts, "translation"),
210             ai_translation=_joined(parts, "ai_translation"), box=rect_poly(_union_of(parts)),
211             line_count=sum(p.get("line_count", 0) for p in parts))
212    if text_raw:
213        a["text_raw"] = text_raw
214    for key in ("skip", "skip_reason", "confirmed_translation", "confirmed_at", "confirmed_via", "layout"):   # layout: the area changed
215        a.pop(key, None)
216    a["merged_by_user"] = True
217    for line in ocr_data.get("lines", []):
218        if line.get("bubble_id") in removed:
219            line["bubble_id"] = bubble_id
220    rest = [x for x in ocr_data["bubbles"] if x["bubble_id"] not in removed and x is not a]
221    rest.insert(first_index, a)
222    ocr_data["bubbles"] = rest
223    if image is not None:
224        _refresh_info(image, ocr_data, a)
225    return {"bubble": a, "removed": removed[0] if len(removed) == 1 else removed, "removed_ids": removed,
226            "order": [p["bubble_id"] for p in parts]}

238# English word that opens a piece -> Polish words the translation of that piece usually starts with
239_LINKS = {
240    "BUT": ("ALE", "LECZ", "JEDNAK", "A"), "AND": ("I", "A", "ORAZ"), "OR": ("ALBO", "LUB", "CZY"),
241    "SO": ("WIĘC", "TO", "DLATEGO", "TAK"), "BECAUSE": ("BO", "PONIEWAŻ", "GDYŻ"), "SINCE": ("SKORO", "PONIEWAŻ", "OD"),
242    "IF": ("JEŚLI", "JEŻELI", "GDY", "GDYBY"), "WHEN": ("GDY", "KIEDY"), "THEN": ("TO", "WTEDY", "POTEM"),
243    "THAT": ("ŻE", "KTÓRY", "KTÓRA", "KTÓRE"), "WHICH": ("KTÓRY", "KTÓRA", "KTÓRE", "CO"), "WHO": ("KTÓRY", "KTÓRA", "KTO"),
244    "ALTHOUGH": ("CHOĆ", "CHOCIAŻ", "MIMO"), "UNTIL": ("AŻ", "DOPÓKI"), "WHILE": ("PODCZAS", "GDY", "A"),
245}
246_CLAUSE_CUT = re.compile(r"(?<=[,;:–—-])\s+|\s+(?=[–—-]\s)")

249def _clause_split(text: str, piece_sources: list[str]) -> list[str] | None:
250    """
251    One-sentence translation, several pieces ("... AT 9:55" | "BUT SINCE THERE IS SOME FREE TIME" ->
252    "..., ALE PONIEWAŻ ..."): cut at clause boundaries (comma, semicolon, dash). A boundary whose next word is a
253    Polish counterpart of the word opening the next source piece (BUT -> ALE) wins; otherwise the boundary whose
254    share of the text is closest to the piece's share of the source. None when there are too few boundaries.
255    """
256    n = len(piece_sources)
257    spots = [m.start() for m in _CLAUSE_CUT.finditer(text)]
258    if len(spots) < n - 1:
259        return None
260    total = max(1, sum(max(1, len(p)) for p in piece_sources))
261    cuts, prev = [], 0
262    for k in range(1, n):
263        share = sum(max(1, len(p)) for p in piece_sources[:k]) / total
264        opener = (piece_sources[k].split() or [""])[0].strip(".,!?\"'").upper()
265        links = _LINKS.get(opener, ())
266        left = [p for p in spots if p > prev and len([q for q in spots if q > p]) >= n - 1 - k]
267        if not left:
268            return None
269        def score(p):
270            nxt = text[p:].lstrip(" ,;:-–—").split(" ", 1)[0].strip(".,!?\"'").upper()
271            return (0 if links and nxt in links else 1, abs(p / len(text) - share))
272        cut = min(left, key=score)
273        cuts.append(cut)
274        prev = cut
275    bounds = [0] + cuts + [len(text)]
276    parts = [text[a:b].strip() for a, b in zip(bounds, bounds[1:])]
277    parts = [p.rstrip(" ,;:–—-").strip() for p in parts[:-1]] + parts[-1:]   # the cut comma does not end a bubble
278    return parts if all(parts) else None

281def split_translation(text: str, piece_sources: list[str]) -> tuple[list[str], str]:
282    """
283    Divide an existing translation between the pieces of a split bubble, in order, WITHOUT asking the translator:
284      "sentences"    the translation has as many sentences as the source pieces together: each piece gets the same
285                     number of sentences as its source text has ("WHAT... IS THIS?!" | "I DON'T KNOW." -> 2 | 1)
286      "proportional" otherwise, with more sentences than pieces: cut between sentences where the share of the text
287                     best matches the share of the source text of each piece
288      "clauses"      fewer sentences than pieces: cut at commas / dashes, preferring a Polish conjunction that matches
289                     the word opening the next source piece (BUT -> ALE), see _clause_split
290      "unsplit"      not even that: the whole translation stays with the first piece, the rest is empty
291    The editor marks "proportional" / "clauses" / "unsplit" pieces "check translation"; Retranslate is up to the user.
292    """
293    n = len(piece_sources)
294    text = (text or "").strip()
295    if n < 2 or not text:
296        return [text] + [""] * max(0, n - 1), "sentences" if n < 2 else "unsplit"
297    sents = _sentences(text)
298    if len(sents) < n:
299        clauses = _clause_split(text, piece_sources)
300        return (clauses, "clauses") if clauses else ([text] + [""] * (n - 1), "unsplit")
301    counts = [len(_sentences(s)) for s in piece_sources]
302    if all(counts) and sum(counts) == len(sents):
303        bounds = [sum(counts[:k]) for k in range(n + 1)]
304        return [" ".join(sents[a:b]) for a, b in zip(bounds, bounds[1:])], "sentences"
305    src = [max(1, len(s)) for s in piece_sources]
306    targets = [sum(src[:k]) / sum(src) for k in range(1, n)]
307    lengths = [len(s) + 1 for s in sents]
308    shares = [sum(lengths[:i]) / sum(lengths) for i in range(1, len(sents))]   # share after sentence i (cut i)
309    cuts, prev = [], 0
310    for k, target in enumerate(targets, start=1):
311        options = range(prev + 1, len(sents) - (n - k) + 1)                     # leave >= 1 sentence per later piece
312        cut = min(options, key=lambda i: abs(shares[i - 1] - target))
313        cuts.append(cut)
314        prev = cut
315    bounds = [0] + cuts + [len(sents)]
316    return [" ".join(sents[a:b]) for a, b in zip(bounds, bounds[1:])], "proportional"

319def bubble_lines(ocr_data: dict, bubble_id: str) -> list[dict]:
320    """
321    The bubble's text lines in reading order (top to bottom, then left to right): [{line_id, box, text}]. The text of
322    a line comes from the page JSON (saved since 2026-11-02) or, for older pages, from the OCR words lying in the line
323    box; "" when neither knows it (split_bubble then reads that piece again).
324    """
325    _find(ocr_data, bubble_id)
326    items = ocr_data.get("items", [])
327    out = []
328    for l in ocr_data.get("lines", []):
329        if l.get("bubble_id") != bubble_id:
330            continue
331        text = (l.get("text") or "").strip()
332        if not text:
333            x1, y1, x2, y2 = bbox_of(l["box"])
334            words = [it for it in items if x1 - 2 <= center(bbox_of(it["box"]))[0] <= x2 + 2
335                     and y1 - 2 <= center(bbox_of(it["box"]))[1] <= y2 + 2]
336            text = " ".join(it["text"] for it in sorted(words, key=lambda it: bbox_of(it["box"])[0])).strip()
337        out.append({"line_id": l["line_id"], "box": l["box"], "text": text})
338    out.sort(key=lambda l: (round(bbox_of(l["box"])[1] / 6), bbox_of(l["box"])[0]))
339    return out

342def _manual_parts(translations, n: int) -> list[str] | None:
343    """The user's own division of the translation (split dialog "I choose the cut"): n strings, or None."""
344    if translations is None:
345        return None
346    if not (isinstance(translations, (list, tuple)) and len(translations) == n and all(isinstance(t, str) for t in translations)):
347        raise BubbleEditError(f"translations must be a list of {n} texts (one per part, in reading order)", 400)
348    return [t.strip() for t in translations]

351def _split_plan(ocr_data: dict, bubble_id: str, new_line_ids, image=None, ocr_region_fn=None, protected=(),
352                translations=None) -> dict:
353    """
354    Pieces of the split (shared by the preview and the real split): kept lines first or in reading order of the page.
355    `translations` = the user's own cut of the translation, one text per piece in reading order (method "manual");
356    the AI draft (ai_translation) is still divided automatically - it stays the baseline the user's text is compared to.
357    """
358    bubble = _find(ocr_data, bubble_id)
359    lines = bubble_lines(ocr_data, bubble_id)
360    if len(lines) < 2:
361        raise BubbleEditError("This bubble has only one text line - nothing to split.", 400)
362    chosen = set(new_line_ids or [])
363    unknown = chosen - {l["line_id"] for l in lines}
364    if unknown:
365        raise BubbleEditError("Some of the chosen lines do not belong to this bubble.", 400)
366    groups = [[l for l in lines if l["line_id"] not in chosen], [l for l in lines if l["line_id"] in chosen]]
367    if not groups[0] or not groups[1]:
368        raise BubbleEditError("Pick some, but not all, lines of the bubble for the new bubble.", 400)
369    groups.sort(key=lambda g: min((round(bbox_of(l["box"])[1] / 6), bbox_of(l["box"])[0]) for l in g))   # source order
370
371    pieces = []
372    for g in groups:
373        union = bbox_of(g[0]["box"])
374        for l in g[1:]:
375            union = union_bbox(union, bbox_of(l["box"]))
376        text = " ".join(l["text"] for l in g if l["text"]).strip()
377        reread = any(not l["text"] for l in g)
378        if reread:
379            if image is None or ocr_region_fn is None:
380                text = ""                                     # preview: the real split reads it
381            else:
382                ih, iw = image.shape[:2]
383                reading = read_region(image, expand(union, 0.15, iw, ih, min_pad=4), ocr_region_fn)
384                text = _clean(reading["text"], protected) if reading else text
385        pieces.append({"line_ids": [l["line_id"] for l in g], "bbox": union, "text": text, "reread": reread})
386
387    sources = [p["text"] or " ".join(l["text"] for l in g) for p, g in zip(pieces, groups)]
388    manual = _manual_parts(translations, len(pieces))
389    if manual is not None:
390        translations, method = manual, "manual"
391    else:
392        translations, method = split_translation(bubble.get("translation", ""), sources)
393    if manual is not None and (bubble.get("translation") or "").strip() == (bubble.get("ai_translation") or "").strip():
394        # the AI draft itself, only cut where the user chose: the parts must not look "edited" (nor become memory
395        # "corrections" when saved) just because the automatic cut of ai_translation landed elsewhere
396        ai_translations = list(manual)
397    else:
398        ai_translations, _ = split_translation(bubble.get("ai_translation", ""), sources)
399    for p, t, a in zip(pieces, translations, ai_translations):

413                 translations=None) -> dict:
414    """

529def set_layout(ocr_data: dict, bubble_id: str, changes: dict, base: dict | None = None) -> dict:
530    """

532      box        [x1, y1, x2, y2] text area in image coordinates; null = automatic (bubble interior)
533      font_size  px; null = automatic (largest size that fits)
534      angle      degrees clockwise, -180..180
535      align      center | left | right
536      stroke     none | thin | thick (outline in the other colour than the letters, for text over artwork)
537      color      black | white (the letters); "color_explicit": true keeps the value even when it is the default
538                 (the letters were white automatically on a dark background and the user wants black)
539      valign     middle | top | bottom

541      bold / italic  true / false (imitated when the family has no such file)

544    before. `base` = the project's lettering style (services/project_prefs.py): font / bold / italic / stroke / color
545    equal to it are not stored (the bubble follows the project), a built-in default that differs from it IS stored

548    base = base or {}
549    bubble = _find(ocr_data, bubble_id)
550    layout = {} if changes.get("reset") else dict(bubble.get("layout") or {})

552        box = changes["box"]
553        if box is None:
554            layout.pop("box", None)
555        else:

557                x1, y1, x2, y2 = (round(float(v), 1) for v in box)

562            if x2 - x1 < 8 or y2 - y1 < 8:

564            layout["box"] = [x1, y1, x2, y2]
565    if "font_size" in changes:
566        if changes["font_size"] in (None, "", 0):
567            layout.pop("font_size", None)
568        else:
569            try:
570                layout["font_size"] = max(6, min(300, int(changes["font_size"])))
571            except (TypeError, ValueError):
572                raise BubbleEditError("font_size must be a number", 400)
573    if "angle" in changes:
574        try:
575            angle = float(changes["angle"] or 0) % 360
576        except (TypeError, ValueError):
577            raise BubbleEditError("angle must be a number", 400)
578        angle = round(angle - 360 if angle > 180 else angle, 1)
579        if angle or bubble_text_rect(ocr_data, bubble):
580            layout["angle"] = angle          # 0 is kept for tilted text: "level" overrides the automatic slant
581        else:
582            layout.pop("angle", None)
583    if "font" in changes:
584        name = changes["font"]
585        if name is not None and not (isinstance(name, str) and len(name) <= 200):
586            raise BubbleEditError("font must be a font family name", 400)
587        name = (name or "").strip()
588        if name == base.get("font", ""):
589            layout.pop("font", None)
590        else:
591            layout["font"] = name            # "" = the Settings font although the project has another one
592    for key in ("bold", "italic"):
593        if key in changes:
594            if bool(changes[key]) == bool(base.get(key, False)):
595                layout.pop(key, None)
596            else:
597                layout[key] = bool(changes[key])
598    for key, allowed, default in (("align", LAYOUT_ALIGNS, "center"), ("stroke", LAYOUT_STROKES, "none"),
599                                  ("valign", LAYOUT_VALIGNS, "middle"), ("color", LAYOUT_COLORS, "black")):
600        if key in changes:
601            if changes[key] not in allowed:
602                raise BubbleEditError(f"{key} must be one of {', '.join(allowed)}", 400)
603            if changes[key] == base.get(key, default) and not (key == "color" and changes.get("color_explicit")):
604                layout.pop(key, None)
605            else:
606                layout[key] = changes[key]
607    if layout:
608        bubble["layout"] = layout
609    else:
610        bubble.pop("layout", None)
611    return {"bubble_id": bubble_id, "layout": layout}

620def auto_order(bubbles: list[dict], mode: str = "ttb") -> list[dict]:
621    """
622    Order bubbles for the translator / the LLM.
623      ttb  top to bottom (then left to right) - what the OCR produces
624      rtl  rows top to bottom, inside a row RIGHT to LEFT   (Japanese manga)
625      ltr  rows top to bottom, inside a row left to right   (western comics, webtoon columns)
626    A "row" is a group of bubbles whose centres are within ~0.8 of a bubble height of each other.
627    LIMITATION: there is no panel detection. On a page whose panels are not in clean rows (a tall panel next to two
628    stacked ones) the heuristic can be wrong - use "Move up / down" in the editor to fix single bubbles.
629    """
630    if mode not in ("ttb", "rtl", "ltr") or len(bubbles) < 2:
631        return list(bubbles)
632    if mode == "ttb":
633        return sorted(bubbles, key=lambda b: (_center_of(b)[1], _center_of(b)[0]))
634
635    heights = sorted(bbox_of(b.get("box") or b["box_coords"])[3] - bbox_of(b.get("box") or b["box_coords"])[1] for b in bubbles)
636    tolerance = 0.8 * heights[len(heights) // 2]
637    ordered = sorted(bubbles, key=lambda b: _center_of(b)[1])
638    rows, current = [], [ordered[0]]
639    for b in ordered[1:]:
640        if abs(_center_of(b)[1] - sum(_center_of(x)[1] for x in current) / len(current)) <= tolerance:
641            current.append(b)
642        else:
643            rows.append(current)
644            current = [b]
645    rows.append(current)
646    out = []
647    for row in rows:
648        out.extend(sorted(row, key=lambda b: _center_of(b)[0], reverse=(mode == "rtl")))
649    return out

285    Divide an existing translation between the pieces of a split bubble, in order, WITHOUT asking the translator:
286      "sentences"    the translation has as many sentences as the source pieces together: each piece gets the same
287                     number of sentences as its source text has ("WHAT... IS THIS?!" | "I DON'T KNOW." -> 2 | 1)
288      "proportional" otherwise, with more sentences than pieces: cut between sentences where the share of the text
289                     best matches the share of the source text of each piece
290      "clauses"      fewer sentences than pieces: cut at commas / dashes, preferring a Polish conjunction that matches
291                     the word opening the next source piece (BUT -> ALE), see _clause_split
292      "unsplit"      not even that: the whole translation stays with the first piece, the rest is empty
293    The editor marks "proportional" / "clauses" / "unsplit" pieces "check translation"; Retranslate is up to the user.
294    """
295    n = len(piece_sources)
296    text = (text or "").strip()
297    if n < 2 or not text:
298        return [text] + [""] * max(0, n - 1), "sentences" if n < 2 else "unsplit"
299    sents = _sentences(text)
300    if len(sents) < n:
301        clauses = _clause_split(text, piece_sources)
302        return (clauses, "clauses") if clauses else ([text] + [""] * (n - 1), "unsplit")
303    counts = [len(_sentences(s)) for s in piece_sources]
304    if all(counts) and sum(counts) == len(sents):
305        bounds = [sum(counts[:k]) for k in range(n + 1)]
306        return [" ".join(sents[a:b]) for a, b in zip(bounds, bounds[1:])], "sentences"
307    src = [max(1, len(s)) for s in piece_sources]
308    targets = [sum(src[:k]) / sum(src) for k in range(1, n)]
309    lengths = [len(s) + 1 for s in sents]
310    shares = [sum(lengths[:i]) / sum(lengths) for i in range(1, len(sents))]   # share after sentence i (cut i)
311    cuts, prev = [], 0
312    for k, target in enumerate(targets, start=1):
313        options = range(prev + 1, len(sents) - (n - k) + 1)                     # leave >= 1 sentence per later piece
314        cut = min(options, key=lambda i: abs(shares[i - 1] - target))
315        cuts.append(cut)
316        prev = cut
317    bounds = [0] + cuts + [len(sents)]
318    return [" ".join(sents[a:b]) for a, b in zip(bounds, bounds[1:])], "proportional"

def auto_order(bubbles: list[dict], mode: str = 'ttb') -> list[dict]: 
622def auto_order(bubbles: list[dict], mode: str = "ttb") -> list[dict]:
623    """
624    Order bubbles for the translator / the LLM.
625      ttb  top to bottom (then left to right) - what the OCR produces
626      rtl  rows top to bottom, inside a row RIGHT to LEFT   (Japanese manga)
627      ltr  rows top to bottom, inside a row left to right   (western comics, webtoon columns)
628    A "row" is a group of bubbles whose centres are within ~0.8 of a bubble height of each other.
629    LIMITATION: there is no panel detection. On a page whose panels are not in clean rows (a tall panel next to two
630    stacked ones) the heuristic can be wrong - use "Move up / down" in the editor to fix single bubbles.
631    """
632    if mode not in ("ttb", "rtl", "ltr") or len(bubbles) < 2:
633        return list(bubbles)
634    if mode == "ttb":
635        return sorted(bubbles, key=lambda b: (_center_of(b)[1], _center_of(b)[0]))
636
637    heights = sorted(bbox_of(b.get("box") or b["box_coords"])[3] - bbox_of(b.get("box") or b["box_coords"])[1] for b in bubbles)
638    tolerance = 0.8 * heights[len(heights) // 2]
639    ordered = sorted(bubbles, key=lambda b: _center_of(b)[1])
640    rows, current = [], [ordered[0]]
641    for b in ordered[1:]:
642        if abs(_center_of(b)[1] - sum(_center_of(x)[1] for x in current) / len(current)) <= tolerance:
643            current.append(b)
644        else:
645            rows.append(current)
646            current = [b]
647    rows.append(current)
648    out = []
649    for row in rows:
650        out.extend(sorted(row, key=lambda b: _center_of(b)[0], reverse=(mode == "rtl")))
651    return out
