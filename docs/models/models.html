 22class Project(Base):
 23    """Represents a translation project containing multiple chapters."""
 24
 25    __tablename__ = "projects"
 26
 27    id = Column(Integer, primary_key=True, index=True)
 28    name = Column(String, unique=True, nullable=False)
 29
 30    # PORTABILITY: the real folder is ALWAYS <workspace root from settings.json>/<project name> (see the
 31    # workspace_path property). The DB column only keeps the folder NAME - an absolute path stored here broke every
 32    # move to another PC / drive (startup treated the moved project as new and crashed on UNIQUE(name), or deleted it).
 33    _stored_workspace = Column("workspace_path", String, nullable=True)
 34    created_at = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc))
 35
 36    source_lang = Column(String, default="en")
 37    target_lang = Column(String, default="pl")
 38
 39    # "active" | "archived". No archive/unarchive endpoint yet - this is the
 40    # schema placeholder for the sidebar's Active/Archived grouping; every
 41    # project defaults to "active" until that workflow is built.
 42    status = Column(String, default="active")
 43
 44    @property
 45    def workspace_path(self) -> str:
 46        """Absolute project folder on THIS machine, derived from the current workspace root + project name."""
 47        return os.path.join(config.WORKSPACE_DIR, self.name)
 48
 49    @workspace_path.setter
 50    def workspace_path(self, value) -> None:
 51        # Accepted so old call sites (Project(name=..., workspace_path=...)) keep working; only the folder name is kept.
 52        # split on BOTH separators: a path written by a Windows machine must reduce to its folder name on Linux/macOS too
 53        self._stored_workspace = re.split(r"[\\/]", str(value).rstrip("\\/"))[-1] if value else self.name
 54
 55    # Relationships
 56    chapters = relationship("Chapter", back_populates="project", cascade="all, delete-orphan")
 57    corrections = relationship("Correction", back_populates="project", cascade="all, delete-orphan")
 58    glossary_terms = relationship("GlossaryTerm", back_populates="project", cascade="all, delete-orphan")

 85        return os.path.join(self.project.workspace_path, "processed", self.number)

130class Correction(Base):
131    """
132    A single confirmed user correction: the original English source text, the
133    AI's initial (zero-shot) proposal, and the final Polish text the user
134    approved.
135
136    This is the core data structure behind the project's correction-memory
137    feature (see services/memory_service.py). The embedding of `source_text`
138    is used for similarity search, letting the translation service "remember"
139    and reuse past corrections for text that reads as similar in the future.
140
141    Corrections are scoped to a single project (not shared globally) to avoid
142    one series' voice/phrasing bleeding into an unrelated one - see README
143    "Known limitations" for the open question of whether this should ever be
144    relaxed (e.g. across volumes of the same series).
145
146    Rows are not overwritten across bubbles: every confirmed bubble becomes a new
147    row, so the memory grows across the project rather than collapsing
148    similar-but-distinct lines into a single entry. Since 2026-11-05 one exception: the SAME bubble saved again
149    (a typo fixed 20 s later) updates its own row while the memory has not used it yet (page_id / bubble_id below,
150    services/memory_service.save_correction) - before, each save added a row, so one bubble left two entries
151    ("WIWERNY" and "WYWERNY") and glossary suggestions counted the same fix twice.
152    """
153
154    __tablename__ = "corrections"
155
156    id = Column(Integer, primary_key=True, index=True)
157    project_id = Column(Integer, ForeignKey("projects.id"), nullable=False)
158
159    source_text = Column(Text, nullable=False)
160    ai_translation = Column(Text)
161    final_translation = Column(Text, nullable=False)
162
163    # Raw float32 embedding bytes for `source_text` (see services/memory_service.py
164    # encode_embedding / decode_embedding). Stored as a plain column rather than in
165    # a dedicated vector DB - at thesis scale, a brute-force in-memory cosine scan
166    # over a few hundred/thousand rows is simpler to reason about and fast enough.
167    embedding = Column(LargeBinary, nullable=False)
168
169    reuse_count = Column(Integer, default=0)
170    created_at = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc))
171
172    # "corrected": the user changed the AI proposal (strongest signal, ranked first at retrieval).
173    # "approved":  the user accepted the AI proposal unchanged - still a confirmed EN->PL pair, i.e. a regular
174    #              translation-memory entry (Moslem 2024 retrieves from the whole TM, not only from edits).
175    # NULL = rows from before this column existed = "corrected" (the old code stored edits only).
176    kind = Column(String, nullable=True, default="corrected")
177
178    # Edited later on the Memory page (inline edit / find & replace), e.g. a term the user changed their mind about
179    # (WIWERN -> WYWERN). final_before_edit keeps the text as it was first saved from the editor, so the evaluation
180    # can tell an in-editor correction from a later memory clean-up. NULL = never edited.
181    edited_at = Column(DateTime(timezone=True), nullable=True)
182    final_before_edit = Column(Text, nullable=True)
183
184    # The bubble this entry was saved from (editor). NULL = rows from before 2026-11-05, imports, or saves without a
185    # page. Not a foreign key on purpose: a deleted page / bubble must not take its memory with it.
186    page_id = Column(Integer, nullable=True)
187    bubble_id = Column(String, nullable=True)
188
189    # Relationships
190    project = relationship("Project", back_populates="corrections")

193class TranslationLog(Base):
194    """
195    Records every translation attempt made for a bubble, including both sides
196    of the zero-shot vs. memory-injected comparison produced by
197    services/translation_service.py.
198
199    This is the raw data the thesis experiment is evaluated from: for bubbles
200    where a correction-memory match was found, two rows are written (one per
201    variant, sharing the same run_id) so the two outputs can later be
202    compared - e.g. via edit distance or chrF - against the translation the
203    user eventually approves for that bubble.
204    """
205
206    __tablename__ = "translation_logs"
207
208    id = Column(Integer, primary_key=True, index=True)
209    page_id = Column(Integer, ForeignKey("pages.id"), nullable=False)
210    bubble_id = Column(String, nullable=False)
211
212    variant = Column(String, nullable=False)  # "zero_shot" | "memory_injected"
213    source_text = Column(Text, nullable=False)
214    output_text = Column(Text)
215
216    matched_correction_id = Column(Integer, ForeignKey("corrections.id"), nullable=True)
217    similarity_score = Column(Float, nullable=True)
218    threshold_used = Column(Float, nullable=True)  # MEMORY_SIMILARITY_THRESHOLD at the time of this match
219
220    model_used = Column(String)
221    key_label = Column(String, nullable=True)  # which key from the provider's pool served this call
222    glossary_terms_used = Column(Text,
223                                 nullable=True)  # JSON list of {"source_term":..., "target_term":...} that fired for this bubble
224    # The EXACT memory material sent with this bubble: {"examples": [{en, pl, similarity, kind, rejected?}, ...]}
225    # (fuzzy lane, best first) or {"exact": "..."}. Logs written before the EN->PL pairs hold {"hints": [pl, ...]}.
226    # matched_correction_id only points at the top-1 correction, which may later be deleted or superseded - this keeps
227    # the experiment reproducible.
228    hints_json = Column(Text, nullable=True)
229    # Which memory lane produced the injected material: "exact" (short phrase, verbatim reuse) | "fuzzy" (embedding
230    # examples) | NULL (no memory). Lets the analysis separate trivial exact hits ("YES" -> "TAK") from fuzzy matches.
231    memory_lane = Column(String, nullable=True)
232    # EVERYTHING sent with / done to this bubble, as one JSON object (see translation_service._injection_record):
233    # memory examples (en/pl/rejected/similarity/kind/correction_id), glossary terms, context bubbles, the terminology
234    # check + post-edit (draft before the fix, which terms were missing), case normalization. For the experiment
235    # write-up: "what exactly did the model see, and what happened afterwards".
236    injection_json = Column(Text, nullable=True)
237    run_id = Column(String, nullable=False)  # groups the zero_shot / memory_injected pair from one translation pass
238
239    created_at = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc))
240
241    # Relationships
242    page = relationship("Page")
243    matched_correction = relationship("Correction")

245class GlossaryTerm(Base):
246    """
247    Deterministic term-base (English -> Polish), separate from the fuzzy
248    embedding-based Correction memory. Correction suggests STYLE for a
249    similar whole segment; GlossaryTerm forces a FIXED translation for one
250    word/phrase, regardless of what sentence it shows up in.
251
252    Matched by word presence (word-boundary regex), not similarity -
253    catches recurring proper nouns/invented terms that will never clear a
254    whole-bubble similarity threshold.
255    """
256    __tablename__ = "glossary_terms"
257
258    id = Column(Integer, primary_key=True, index=True)
259    project_id = Column(Integer, ForeignKey("projects.id"), nullable=False)
260
261    source_term = Column(String, nullable=False)
262    target_term = Column(String, nullable=False)
263
264    auto_detected = Column(Boolean, default=False)
265    occurrences = Column(Integer, default=1)
266    created_at = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc))
267    # Where the term came from (since 2026-11-11; NULL on older rows - there auto_detected is all that is known):
268    # "manual" typed by the user, "scan" repeated-fix pattern found in the corrections, "ai" extracted by the LLM,
269    # "import" from a glossary file. Every suggestion was ACCEPTED by the user; proposed_term = the Polish term as it
270    # was suggested, so target_term != proposed_term means the user changed it before accepting.
271    origin = Column(String, nullable=True)
272    proposed_term = Column(String, nullable=True)
273
274    project = relationship("Project", back_populates="glossary_terms")

A single confirmed user correction: the original English source text, the AI's initial (zero-shot) proposal, and the final Polish text the user approved.

This is the core data structure behind the project's correction-memory feature (see services/memory_service.py). The embedding of source_text is used for similarity search, letting the translation service "remember" and reuse past corrections for text that reads as similar in the future.

Corrections are scoped to a single project (not shared globally) to avoid one series' voice/phrasing bleeding into an unrelated one - see README "Known limitations" for the open question of whether this should ever be relaxed (e.g. across volumes of the same series).

Rows are not overwritten across bubbles: every confirmed bubble becomes a new row, so the memory grows across the project rather than collapsing similar-but-distinct lines into a single entry. Since 2026-11-05 one exception: the SAME bubble saved again (a typo fixed 20 s later) updates its own row while the memory has not used it yet (page_id / bubble_id below, services/memory_service.save_correction) - before, each save added a row, so one bubble left two entries ("WIWERNY" and "WYWERNY") and glossary suggestions counted the same fix twice.
